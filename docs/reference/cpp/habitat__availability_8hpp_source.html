60inline std::array<float, kNumZoneId> habitat_potential(
61 const HabitatFeatures& f) {
62 std::array<float, kNumZoneId> h{};
63 h[static_cast<size_t>(ZoneId::JRC_PERSISTENT)] =
64 std::clamp(f.jrc_persistent, 0.0f, 1.0f);
65 // Urban persistent supply — extends the lease.
66 h[static_cast<size_t>(ZoneId::URBAN_PERSISTENT)] =
67 std::clamp(static_cast<float>(
68 load_config().urban.beta_urban) *
69 f.lc_built,
70 0.0f, 1.0f);
71 h[static_cast<size_t>(ZoneId::WETLAND)] =
72 std::clamp(f.lc_wetland, 0.0f, 1.0f);
73 // Open pluvial: open/low-veg + water + rain pools on the built surface…
74 // (documented rule; does NOT double-count man-made supply: pools go
75 // under a distinct habitat type and a distinct temporal dynamic.)
76 h[static_cast<size_t>(ZoneId::OPEN_PLUVIAL)] =
77 std::clamp(f.lc_open_plus_water +
78 static_cast<float>(load_config()
79 .urban.rain_pools_built_damage) *
80 f.lc_built,
81 0.0f, 1.0f);
82 h[static_cast<size_t>(ZoneId::CROP)] =
83 std::clamp(f.lc_crop, 0.0f, 1.0f);
84 h[static_cast<size_t>(ZoneId::FOREST)] = 0.0f;
85 return h;
86}

102inline float update_z_step(float z, float rain_mm, float delta_days) {
103 const auto& pluvial = load_config().pluvial;
104 const float z_dry =
105 z * std::exp(-delta_days / static_cast<float>(pluvial.dry_memory_days));
106 const float fill =
107 1.0f - std::exp(-std::max(0.0f, rain_mm) /
108 static_cast<float>(pluvial.rain_scale_mm));
109 return std::clamp(z_dry + (1.0f - z_dry) * fill, 0.0f, 1.0f);
110}

167inline float species_accuracy(const std::array<float, kNumZoneId>& h,
168 const std::array<float, kNumZoneId>& w,
169 float z_pluvial) {
170 // Persistent/pluvial split comes from habitat_types.dynamics (names),
171 // precomputed as a ZoneId bitset at load time (no positional rule).
172 const auto& persistent_zone = load_config().habitat_types.persistent_zone;
173 float s = 0.0f;
174 for (size_t g = 0; g < kNumZoneId; ++g) {
175 if (w[g] <= 0.0f || h[g] <= 0.0f) continue;
176 const float availability =
177 persistent_zone[g]
178 ? static_cast<float>(
179 load_config().capacity.persistent_availability)
180 : std::clamp(z_pluvial, 0.0f, 1.0f);
181 s += w[g] * h[g] * availability;
182 }
183 return std::clamp(s, 0.0f, 1.0f);
184}

196inline double cell_area_km2(double lat_deg, double cell_size_m) {
197 // Equirectangular approximation, adequate at the scale of a single cell:
198 // area = dx * dy where dx ≈ cell_size * cos(lat).
199 constexpr double kDegToRad = 3.14159265358979323846 / 180.0;
200 const double dx_m = cell_size_m * std::cos(lat_deg * kDegToRad);
201 return (dx_m * cell_size_m) / 1.0e6;
202}
