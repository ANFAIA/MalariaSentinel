76inline std::array<float, kNumZoneId> habitat_potential(
77 const HabitatFeatures& f) {
78 std::array<float, kNumZoneId> h{};
79 h[static_cast<size_t>(ZoneId::JRC_PERSISTENT)] =
80 std::clamp(f.jrc_persistent, 0.0f, 1.0f);
81 // Urban persistent supply — extends the lease.
82 h[static_cast<size_t>(ZoneId::URBAN_PERSISTENT)] =
83 std::clamp(static_cast<float>(
84 load_config().urban.beta_urban) *
85 f.lc_built,
86 0.0f, 1.0f);
87 h[static_cast<size_t>(ZoneId::WETLAND)] =
88 std::clamp(f.lc_wetland, 0.0f, 1.0f);
89 // Open pluvial: open/low-veg + water + rain pools on the built surface…
90 // (documented rule; does NOT double-count man-made supply: pools go
91 // under a distinct habitat type and a distinct temporal dynamic.)
92 h[static_cast<size_t>(ZoneId::OPEN_PLUVIAL)] =
93 std::clamp(f.lc_open_plus_water +
94 static_cast<float>(load_config()
95 .urban.rain_pools_built_damage) *
96 f.lc_built,
97 0.0f, 1.0f);
98 h[static_cast<size_t>(ZoneId::CROP)] =
99 std::clamp(f.lc_crop, 0.0f, 1.0f);
100 h[static_cast<size_t>(ZoneId::FOREST)] = 0.0f;
101 return h;
102}

118inline float update_z_step(float z, float rain_mm, float delta_days) {
119 const auto& pluvial = load_config().pluvial;
120 const float z_dry =
121 z * std::exp(-delta_days / static_cast<float>(pluvial.dry_memory_days));
122 const float fill =
123 1.0f - std::exp(-std::max(0.0f, rain_mm) /
124 static_cast<float>(pluvial.rain_scale_mm));
125 return std::clamp(z_dry + (1.0f - z_dry) * fill, 0.0f, 1.0f);
126}

186inline float species_accuracy(const std::array<float, kNumZoneId>& h,
187 const std::array<float, kNumZoneId>& w,
188 float z_pluvial) {
189 // Persistent/pluvial split comes from habitat_types.dynamics (names),
190 // precomputed as a ZoneId bitset at load time (no positional rule).
191 const auto& persistent_zone = load_config().habitat_types.persistent_zone;
192 float s = 0.0f;
193 for (size_t g = 0; g < kNumZoneId; ++g) {
194 if (w[g] <= 0.0f || h[g] <= 0.0f) continue;
195 const float availability =
196 persistent_zone[g]
197 ? static_cast<float>(
198 load_config().capacity.persistent_availability)
199 : std::clamp(z_pluvial, 0.0f, 1.0f);
200 s += w[g] * h[g] * availability;
201 }
202 return std::clamp(s, 0.0f, 1.0f);
203}

215inline double cell_area_km2(double lat_deg, double cell_size_m) {
216 // Equirectangular approximation, adequate at the scale of a single cell:
217 // area = dx * dy where dx ≈ cell_size * cos(lat).
218 constexpr double kDegToRad = 3.14159265358979323846 / 180.0;
219 const double dx_m = cell_size_m * std::cos(lat_deg * kDegToRad);
220 return (dx_m * cell_size_m) / 1.0e6;
221}
