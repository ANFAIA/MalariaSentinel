285inline int32_t AOI::cells_per_side() const noexcept {
286 const double centroid_lat_rad = ((north + south) * 0.5) * (kPi / 180.0);
287 const double width_m = (east - west) * load_config().geodesy.metres_per_deg_lat * std::cos(centroid_lat_rad);
288 const double cells_d = width_m / static_cast<double>(resolution_m);
289 int32_t cells = static_cast<int32_t>(std::ceil(cells_d));
290 if (cells < 1) cells = 1;
291 return cells;
292}
