365inline int32_t AOI::cells_per_side() const noexcept {
366 const double centroid_lat_rad = ((north + south) * 0.5) * (kPi / 180.0);
367 const double width_m = (east - west) * load_config().geodesy.metres_per_deg_lat * std::cos(centroid_lat_rad);
368 const double cells_d = width_m / static_cast<double>(resolution_m);
369 int32_t cells = static_cast<int32_t>(std::ceil(cells_d));
370 if (cells < 1) cells = 1;
371 return cells;
372}
