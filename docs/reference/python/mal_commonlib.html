 66    """Area of Interest — region-agnostic from M1.
 67
 68    The AOI is a typed parameter; the data layer and the ABM consume
 69    bbox + CRS + scale + slug. No hardcoded Ghana logic.
 70
 71    Attributes:
 72        slug: stable identifier used in file naming (e.g. "ghana", "mozambique").
 73              Lowercase, [a-z0-9-], no leading/trailing hyphens, no consecutive hyphens.
 74        name: human-readable name (e.g. "Ghana NMCP AOI").
 75        bbox: (west, south, east, north) in CRS units. For EPSG:4326, west<east and south<north.
 76        crs: EPSG code or WKT. Must be parseable by pyproj.CRS.from_user_input.
 77        resolution_m: cell size in metres on the ground. For EPSG:4326, used to
 78                      derive degree-per-cell via the AOI centroid latitude.
 79        scale: REGIONAL/NATIONAL/CONTINENTAL — see ``Scale`` enum.
 80        gadm_id: optional GADM identifier (e.g. "GHA" for Ghana, "GHA.1_1" for region).
 81                 When set, ``from_gadm()`` can be used to populate bbox from GADM boundaries.
 82        iso3: ISO 3166-1 alpha-3 country code (e.g. "GHA"). Required for country-scoped
 83              data loaders (WorldPop, GADM, livestock); raises ValueError if absent when
 84              needed. Always lowercase, exactly 3 letters.
 85    """

102    def _slug_format(cls, v: str) -> str:
103        """Reject slugs that are not lowercase alphanumeric words joined by single hyphens."""
104        if not re.fullmatch(r"[a-z0-9]+(-[a-z0-9]+)*", v):
105            raise ValueError(
106                f"slug must be lowercase [a-z0-9-], no leading/trailing/consecutive hyphens; got {v!r}"
107            )
108        return v

110    @field_validator("iso3")
111    @classmethod
112    def _iso3_format(cls, v: str | None) -> str | None:
113        """Normalise an ISO 3166-1 alpha-3 code to uppercase and reject codes that are not three letters."""
114        if v is None:
115            return v
116        v = v.strip().upper()
117        if not re.fullmatch(r"[A-Z]{3}", v):
118            raise ValueError(
119                f"iso3 must be 3 letters (ISO 3166-1 alpha-3); got {v!r}"
120            )
121        return v

123    @field_validator("bbox")
124    @classmethod
125    def _bbox_order(cls, v: tuple[float, float, float, float]) -> tuple[float, float, float, float]:
126        """Reject a bbox whose west/east or south/north edges are not strictly increasing."""
127        w, s, e, n = v
128        if not (w < e and s < n):
129            raise ValueError(f"bbox must satisfy west<east and south<north; got {v!r}")
130        return v

154    @property
155    def _centroid_lat(self) -> float:
156        """Centroid latitude in degrees (geographic CRS) or None for projected."""
157        _, s, _, n = self.bbox
158        return 0.5 * (s + n)
159
160    def _m_per_unit(self) -> tuple[float, float]:
161        """Approximate (m_per_unit_x, m_per_unit_y) at the AOI centroid.
162
163        Geographic CRS → great-circle metres per degree. Projected CRS → CRS linear units.
164        """
165        if self._is_geographic:
166            lat_rad = math.radians(self._centroid_lat)
167            m_per_deg_lat = (math.pi * _EARTH_R) / 180.0
168            m_per_deg_lon = m_per_deg_lat * math.cos(lat_rad)
169            return m_per_deg_lon, m_per_deg_lat
170        # Projected: assume linear units are metres (UTM, etc.). The CRS conversion factor
171        # is reported by pyproj; fall back to 1.0 if unknown.
172        cf = self.crs_obj.coordinate_system
173        try:
174            unit_m = float(cf.unit_conversion_factor) if cf is not None and cf.unit_conversion_factor else 1.0
175        except (AttributeError, TypeError):
176            unit_m = 1.0
177        return unit_m, unit_m

181        """East-west extent of the bounding box in metres."""
182        w, _, e, _ = self.bbox

307    @classmethod
308    def from_gadm(
309        cls,
310        gadm_id: str, crs: str, slug: str, resolution_m: int,
311        *,
312        gadm_path: Path | str | None = None,
313        name: str | None = None,
314        scale: Scale = Scale.REGIONAL,
315        iso3: str | None = None,
316    ) -> "AOI":
317        """Load an AOI from a GADM file.
318
319        Looks up ``gadm_id`` in the GADM shapefile and computes the bbox from the
320        matching geometry. The GADM path resolves in this order:
321            1. explicit ``gadm_path`` argument
322            2. ``MAL_GADM_PATH`` environment variable
323            3. otherwise: ``FileNotFoundError`` with a clear message.
324
325        The id is matched against any ``GID_<n>`` column (GADM-0..GADM-5).
326        For country-level ids (e.g. "GHA"), pass ``iso3="GHA"`` so country-scoped
327        data loaders work; if omitted, it is inferred from the first three chars
328        of the GADM id (uppercased).
329        """
330        if gadm_path is None:
331            env = os.environ.get("MAL_GADM_PATH")
332            if not env:
333                raise FileNotFoundError(
334                    "GADM path not provided: pass gadm_path=... or set MAL_GADM_PATH"
335                )
336            gadm_path = env
337
338        gdf = gpd.read_file(str(gadm_path))
339        gid_col = None
340        for col in gdf.columns:
341            if col.startswith("GID_") and col[4:].isdigit():
342                if gid_col is None or int(col[4:]) > int(gid_col[4:]):
343                    gid_col = col
344        if gid_col is None:
345            raise ValueError(
346                f"No GID_<n> column found in {gadm_path}; columns={list(gdf.columns)}"
347            )
348
349        match = gdf[gdf[gid_col] == gadm_id]
350        if match.empty:
351            raise KeyError(
352                f"gadm_id {gadm_id!r} not found in {gadm_path} (column {gid_col})"
353            )
354
355        geom = match.geometry
356        # Reproject if needed so the bbox is in the requested CRS.
357        if str(gdf.crs) != str(pyproj.CRS.from_user_input(crs)):
358            geom = geom.to_crs(crs)
359        minx, miny, maxx, maxy = geom.total_bounds
360
361        # Infer ISO-3 from the first three chars of the GADM id when not given.
362        if iso3 is None and re.fullmatch(r"[A-Z]{3}", gadm_id[:3]):
363            iso3 = gadm_id[:3]
364
365        return cls(
366            slug=slug,
367            name=name if name is not None else f"{slug} ({gadm_id})",
368            bbox=(float(minx), float(miny), float(maxx), float(maxy)),
369            crs=crs,
370            resolution_m=resolution_m,
371            scale=scale,
372            gadm_id=gadm_id,
373            iso3=iso3,
374        )
375
376    def require_iso3(self) -> str:
377        """Return ``iso3`` or raise ``ValueError`` if missing.
378
379        Country-scoped data loaders (WorldPop, GADM, livestock) need the ISO-3
380        country code to build download URLs. Call this at the top of any
381        loader that requires a country-level identifier.
382        """
383        if not self.iso3:
384            raise ValueError(
385                f"AOI {self.slug!r} has no iso3; pass iso3='XXX' to "
386                f"from_bbox/from_slug/from_gadm or set it in aois.yaml"
387            )
388        return self.iso3
