mal_commonlib.data.loaders

Data loaders for the env channels, habitat statics, hosts and mobility.

Plugin modules — each exports a public load_* function and a DOWNLOADER dict consumed by mal_core.download.registry (pkgutil scan, no manual list). Contract v3.0: the four land-cover statics (lc_built/lc_wetland/lc_open_plus_water/lc_crop) come from Google Dynamic World V1 (per-year modal label composites on Earth Engine); the persistent-water plane stays JRC GSW. The WorldCover/DEM/annual-ERA5 loaders are retired (git history).

265def load_jrc_gsw_habitat_features(
266    aoi: AOI,
267    *,
268    year: int = 2021,
269    cache_dir: pathlib.Path | None = None,
270) -> xr.Dataset:
271    """Load the contract v3.0 static habitat features derived from the JRC
272    Global Surface Water (GSW) 30 m ``occurrence`` band.
273
274    Single channel retained from the retired binary water/seasonal split
275    (plan docs/plans/in-process/redesign/01-retirada-hidrologia-dem-escorrentia.md
276    §6): the per-cell FRACTION of water pixels with occurrence >= 80 %,
277    coastline-masking excluded, as the only static surface-water signal the
278    redesign consumes (``jrc_persistent`` feature column). The old
279    threshold-80 binary mask (a whole-country binary raster with ~5 525
280    cells, useless for a relative habitat index) is NOT produced anymore;
281    the fraction IS the feature.
282
283    The loader goes through the Microsoft Planetary Computer STAC catalog
284    (``jrc-gsw`` collection) and reads only the AOI bbox window from
285    each intersecting tile.
286
287    Args:
288        aoi: the AOI.
289        year: JRC GSW product year (snapshot selector). The annual
290            ``occurrence`` band is available for 1984-2021 inclusive.
291            Default is 2021 (the latest available year).
292        cache_dir: optional local cache (currently unused — the STAC
293            signed URLs stream directly).
294
295    Returns:
296        ``xr.Dataset`` with one variable ``jrc_persistent`` of dims
297        (y, x), dtype float32, CRS = ``aoi.crs``, in [0, 1];
298        ``NODATA_SENTINEL()`` for NoData.
299    """
300    if not (_MIN_YEAR <= int(year) <= _MAX_YEAR):
301        raise ValueError(
302            f"JRC GSW year must be in [{_MIN_YEAR}, {_MAX_YEAR}]; got {year}"
303        )
304
305    bbox = _aoi_bbox_wgs84(aoi)
306    if not all(np.isfinite(b) for b in bbox):
307        raise ValueError(f"AOI bbox has non-finite values: {bbox}")
308
309    # The feature needs NO 80 % binarisation: the per-cell mean of the
310    # binary "occurrence >= threshold" mask directly IS the fraction of
311    # the cell covered by >=80 %-permanent water pixels. The retired code
312    # multiplied the resulting fraction into a separate 0.8 binary step —
313    # dropped; the mean at threshold_pct = PERMANENT_THRESHOLD_PCT.
314    arr, profile = _load_jrc_gsw_pc(
315        aoi, int(year), int(PERMANENT_THRESHOLD_PCT)
316    )
317
318    values = arr.astype(np.float32)
319    values = np.where(values == _NODATA_OUT_SCALAR, values,
320                      np.clip(values, 0.0, 1.0))
321
322    out = xr.DataArray(
323        values,
324        dims=("y", "x"),
325        name="jrc_persistent",
326        attrs={
327            "long_name": "JRC GSW fraction of >=80%-occurrence water pixels",
328            "source": f"JRC Global Surface Water {year} (Planetary Computer STAC)",
329            "units": "fraction of cell area [0, 1]",
330            "threshold_pct": int(PERMANENT_THRESHOLD_PCT),
331            "asset": _PC_ASSET,
332            "nodata": _NODATA_OUT_SCALAR,
333        },
334    )
335    out.rio.write_crs(profile["crs"], inplace=True)
336    if str(profile["crs"]) != str(aoi.crs_obj):
337        ref = _make_reference_grid(aoi)
338        out = out.rio.reproject_match(ref, resampling=Resampling.average)
339        out = out.astype(np.float32)
340    ds = out.to_dataset(name="jrc_persistent")
341    ds.rio.write_crs(aoi.crs, inplace=True)
342    ds.rio.write_transform(from_bounds(*aoi.bbox, *aoi.cells_per_side()), inplace=True)
343    ds["jrc_persistent"].rio.write_nodata(_NODATA_OUT_SCALAR, inplace=True)
344    ds.attrs = {
345        "contract": "habitat-v3.0-statics",
346        "source": f"JRC Global Surface Water {year} (Planetary Computer STAC)",
347        "asset": _PC_ASSET,
348        "year_used": int(profile["jrc_gsw_year"]),
349        "nodata": _NODATA_OUT_SCALAR,
350    }
351    return ds

347    """Per-year Dynamic World modal-label fractions on the AOI grid.
348
349    Args:
350        aoi: the AOI.
351        years: run years to composite (one modal-label composite each).
352            The returned dataset has a ``time`` dim with one frame per year
353            (Jan 1 of that year); the ingest averages the frames into the
354            single static plane of the contract.
355        cache_dir: local cache root for the modal-label tiles (default
356            ``~/.cache/mal_commonlib/dynamic_world``).
357
358    Returns:
359        ``xr.Dataset`` dims (time, y, x) with variables ``lc_built``,
360        ``lc_wetland``, ``lc_crop``, ``lc_open_plus_water`` (float32,

362    """
363    if not years:
364        raise ValueError("load_dw_class_fractions needs at least one year")
365    if min(years) < MIN_YEAR:
366        raise ValueError(
367            f"Dynamic World record starts {MIN_YEAR}-06-27; got {min(years)}"
368        )
369
370    bbox = _aoi_bbox_wgs84(aoi)
371    if not all(np.isfinite(b) for b in bbox):
372        raise ValueError(f"AOI bbox has non-finite values: {bbox}")
373
374    cache = _ensure_cache_dir(cache_dir)
375    H, W = aoi.cells_per_side()
376    ref_transform = from_bounds(*aoi.bbox, int(W), int(H))
377
378    frames: list[xr.Dataset] = []
379    for year in sorted(set(int(y) for y in years)):
380        data, _ = _year_fraction_planes(
381            year, bbox, ref_transform, aoi.crs_obj, (int(H), int(W)), cache
382        )
383        ds = xr.Dataset(
384            {name: xr.DataArray(arr, dims=("y", "x"), name=name)
385             for name, arr in data.items()},
386            attrs={
387                "contract": "habitat-v3.0-statics",
388                "source": f"Dynamic World V1 modal label, year {year} (GEE)",
389                "resolution_m": _REQUEST_SCALE_M,
390                # netCDF attrs cannot hold dicts — serialize as JSON.
391                "class_map": json.dumps(
392                    {k: list(v) for k, v in FEATURE_CLASSES.items()}
393                ),
394                "attribution": ATTRIBUTION.replace("[Year]", str(year)),
395                "nodata": NODATA_OUT,
396            },
397        )
398        ds = ds.assign_coords(time=np.datetime64(f"{year}-01-01", "D"))
399        ds.rio.write_crs(aoi.crs_obj, inplace=True)
400        ds.rio.write_transform(
401            from_bounds(*aoi.bbox, *aoi.cells_per_side()), inplace=True
402        )
403        for name in data:
404            ds[name].rio.write_nodata(NODATA_OUT, inplace=True)
405        frames.append(ds)
406

408    out.attrs.update(frames[0].attrs)
409    out.attrs["source"] = (
410        "Dynamic World V1 modal label per year, averaged over "
411        f"{[int(y) for y in sorted(set(int(y) for y in years))]} (GEE)"
412    )
413    return out

416def load_dw_label_mode(
417    aoi: "AOI",
418    *,
419    year: int,
420    cache_dir: Path | None = None,
421) -> xr.DataArray:
422    """Dominant-class Dynamic World raster on the AOI grid (shared utility).
423
424    Same role the retired ``load_worldcover_landcover`` had for independent
425    consumers (the wildlife host proxy loader maps the label raster through
426    a suitability table). NOT part of the habitat contract — the habitat
427    ingest consumes ``load_dw_class_fractions`` only.
428
429    Returns:
430        xr.DataArray dims (y, x), dtype int32, CRS = aoi.crs, values are
431        Dynamic World label codes 0-8. ``ingest.nodata_sentinel_int`` NoData.
432    """
433    bbox = _aoi_bbox_wgs84(aoi)
434    if not all(np.isfinite(b) for b in bbox):
435        raise ValueError(f"AOI bbox has non-finite values: {bbox}")
436
437    cache = _ensure_cache_dir(cache_dir)
438    H, W = aoi.cells_per_side()
439    ref_transform = from_bounds(*aoi.bbox, int(W), int(H))
440
441    vote = np.full((H, W), -1, dtype=np.int32)
442    votes_acc = np.zeros((H, W, 9), dtype=np.float32)
443    valid = np.zeros((H, W), dtype=np.float32)
444    for path in _year_mode_label_tifs(year, bbox, cache):
445        raw, win_transform, src_crs = _read_tile(path)
446        h, w = raw.shape
447        tile_valid = np.zeros((H, W), dtype=np.float32)
448        reproject(
449            source=np.ones((h, w), dtype=np.float32),
450            destination=tile_valid,
451            src_transform=win_transform,
452            src_crs=src_crs,
453            dst_transform=ref_transform,
454            dst_crs=aoi.crs_obj,
455            resampling=Resampling.nearest,
456        )
457        for code in range(9):
458            layer = (raw == code).astype(np.float32)
459            plane = np.zeros((H, W), dtype=np.float32)
460            reproject(
461                source=layer,
462                destination=plane,
463                src_transform=win_transform,
464                src_crs=src_crs,
465                dst_transform=ref_transform,
466                dst_crs=aoi.crs_obj,
467                resampling=Resampling.average,
468            )
469            votes_acc[:, :, code] = np.maximum(
470                votes_acc[:, :, code], plane * (tile_valid > 0)
471            )
472        valid = np.maximum(valid, tile_valid)
473
474    covered = valid > 0
475    if covered.any():
476        vote[covered] = np.argmax(votes_acc[covered], axis=-1).astype(np.int32)
477    vote[~covered] = NODATA_OUT_INT
478
479    da = xr.DataArray(vote, dims=("y", "x"), name="label")
480    da.rio.write_crs(aoi.crs_obj, inplace=True)
481    da.rio.write_transform(
482        from_bounds(*aoi.bbox, *aoi.cells_per_side()), inplace=True
483    )
484    da.rio.write_nodata(NODATA_OUT_INT, inplace=True)
485    da.attrs["source"] = f"Dynamic World V1 modal label {year} (GEE)"
486    da.attrs["attribution"] = ATTRIBUTION.replace("[Year]", str(year))
487    return da

251def load_chirps_rainfall_daily(
252    aoi: AOI,
253    *,
254    years: Sequence[int],
255    months: Sequence[int] | None = None,
256    cache_dir: pathlib.Path | None = None,
257    _fetch_daily: Callable[[int, int, int], xr.DataArray] | None = None,
258) -> xr.DataArray:
259    """Load CHIRPS v2.0 daily precipitation as a 3-D (time, y, x) array.
260
261    This is the canonical CHIRPS interface: per-day values written into a
262    daily NetCDF env file. The historical monthly-total loader
263    (``load_chirps_rainfall``) has been deleted; daily rainfall is what the
264    ABM's pluvial availability response needs. Each day's values are
265    reprojected to the AOI's grid independently.
266
267    Args:
268        aoi: the AOI (bbox, CRS, resolution_m, slug).
269        years: sequence of years to load.
270        months: sequence of 1-indexed months. None = all 12.
271        cache_dir: optional local cache for downloaded GeoTIFFs.
272        _fetch_daily: testing hook (kw-only).
273
274    Returns:
275        xr.DataArray with dims (time, y, x), dtype float32, CRS = aoi.crs.
276        Values in mm/day. Time coordinate uses ``np.datetime64`` for each
277        day across all requested year x month combinations.
278    """
279    if months is None:
280        months = list(range(1, 13))
281
282    if _fetch_daily is None:
283        cdir = cache_dir if cache_dir is not None else _default_cache_dir()
284
285        def _default_fetch(y: int, m: int, d: int) -> xr.DataArray:
286            """Download one CHIRPS day and clip it to the AOI, caching the clipped GeoTIFF."""
287            url = _daily_url_for(y, m, d)
288            dest = _cache_path_aoi(cdir, aoi.slug, y, m, d)
289            return _download_and_clip_to_aoi(url, aoi, dest)
290
291        fetch = _default_fetch
292    else:
293        fetch = _fetch_daily
294
295    daily_arrays: list[xr.DataArray] = []
296    time_coords: list[np.datetime64] = []
297
298    for year in sorted(years):
299        for month in sorted(months):
300            if not (1 <= month <= 12):
301                raise ValueError(f"month must be in 1..12; got {month}")
302            if year < 1981:
303                raise ValueError(f"CHIRPS starts in 1981; got year={year}")
304
305            days = _days_in_month(year, month)
306            for d in days:
307                raw = fetch(year, month, d)
308                reproj = _reproject_to_aoi_grid(raw, aoi)
309                nodata_mask = (~np.isfinite(reproj)) | (reproj == CHIRPS_NODATA)
310                reproj = np.where(
311                    nodata_mask, np.float32(CHIRPS_NODATA), reproj.astype(np.float32)
312                )
313                da = xr.DataArray(
314                    reproj.astype(np.float32)[np.newaxis, :, :],  # (1, y, x)
315                    dims=("time", "y", "x"),
316                )
317                daily_arrays.append(da)
318                time_coords.append(
319                    np.datetime64(f"{year:04d}-{month:02d}-{d:02d}")
320                )
321
322    stacked = xr.concat(daily_arrays, dim="time")
323    stacked = stacked.assign_coords(time=time_coords)
324    stacked.attrs.update({
325        "long_name": "CHIRPS v2.0 daily precipitation",

250def distance_to_land_m(
251    land_frac: np.ndarray, resolution_m: float,
252    water_floor: float | None = None,
253) -> np.ndarray:
254    """Euclidean distance (m) from each cell to the nearest land cell.
255
256    Land = ``land_frac >= water_floor`` (default ``ingest.sea_water_floor``).
257    Computed on the AOI grid with ``scipy.ndimage.distance_transform_edt``;
258    one cell unit equals ``resolution_m``.
259    """
260    lf = np.asarray(land_frac, dtype=np.float64)
261    floor = default_sea_water_floor() if water_floor is None else float(water_floor)
262    land = lf >= floor
263    if not land.any():
264        return np.full(lf.shape, np.inf, dtype=np.float64)
265    return ndimage.distance_transform_edt(~land) * float(resolution_m)

169def load_wildlife_host_proxy(
170    aoi: AOI | str,
171    *,
172    year: int = 2021,
173    cache_dir: pathlib.Path | None = None,
174) -> xr.DataArray:
175    """Load wildlife host proxy suitability for the AOI.
176
177    Computes a [0,1] suitability score from ESA WorldCover habitat
178    classes, JRC GSW water presence, and building remoteness.
179
180    Args:
181        aoi: the AOI (bbox, CRS, resolution_m, slug).
182        year: Dynamic World composite year (2015 → today).
183        cache_dir: optional local cache for downloaded data.
184
185    Returns:
186        xr.DataArray with dims (y, x), dtype float32, CRS = aoi.crs.
187        Values in [0, 1]. ``NODATA_SENTINEL()`` for NoData.
188    """
189    if isinstance(aoi, str):
190        aoi = AOI.from_slug(aoi)
191
192    # Model parameters are read before any I/O: a missing/invalid key must be
193    # a loud error, never a silently neutral component.
194    suitability = _class_suitability()
195    weights = _component_weights()
196
197    H, W = aoi.cells_per_side()
198    H, W = int(H), int(W)
199
200    # --- 1. Habitat suitability from Dynamic World --------------------
201    try:
202        habitat = _load_dw_habitat_pc(aoi, year, suitability=suitability)
203        log.info("wildlife: Dynamic World habitat loaded successfully")
204    except Exception as exc:
205        log.warning("wildlife: Dynamic World failed (%s), using neutral 0.5", exc)
206        habitat = np.full((H, W), 0.5, dtype=np.float32)
207
208    # --- 2. Water proximity from JRC GSW -----------------------------
209    try:
210        from mal_commonlib.data.loaders.jrc_gsw import (
211            load_jrc_gsw_habitat_features,
212        )
213
214        jrc = load_jrc_gsw_habitat_features(aoi, cache_dir=cache_dir)
215        water_frac = jrc["jrc_persistent"].values.astype(np.float32)
216        log.info("wildlife: JRC persistent-water surface loaded successfully")
217    except Exception as exc:
218        log.warning("wildlife: JRC GSW failed (%s), using neutral 0.5", exc)
219        water_frac = np.full((H, W), 0.5 / 3.0, dtype=np.float32)
220
221    water = _water_presence(water_frac)
222
223    # --- 3. Remoteness from buildings ---------------------------------
224    try:
225        from mal_commonlib.data.loaders.buildings import load_buildings_fraction
226
227        bld_da = load_buildings_fraction(aoi, cache_dir=cache_dir)
228        bld_frac = bld_da.values.astype(np.float32)
229        # Replace nodata with 0 (no buildings → remote)
230        bld_frac[bld_frac == _NODATA_OUT_SCALAR] = 0.0
231        log.info("wildlife: Buildings loaded successfully")
232    except Exception as exc:
233        log.warning("wildlife: Buildings failed (%s), assuming remote", exc)
234        bld_frac = np.zeros((H, W), dtype=np.float32)
235
236    remote = _remoteness(bld_frac)
237
238    # --- 4. Compute composite ----------------------------------------
239    proxy = (weights["habitat"] * habitat
240             + weights["water"] * water
241             + weights["remoteness"] * remote)
242    proxy = np.clip(proxy, 0.0, 1.0).astype(np.float32)
243
244    # Propagate nodata: cells where WorldCover had no data
245    nodata_mask = habitat == _NODATA_OUT_SCALAR
246    proxy[nodata_mask] = _NODATA_OUT_SCALAR
247
248    # --- 5. Wrap and return -------------------------------------------
249    da = xr.DataArray(
250        proxy,
251        dims=("y", "x"),
252        name="wildlife_host_proxy",
253        attrs={
254            "long_name": "Wildlife host proxy suitability",
