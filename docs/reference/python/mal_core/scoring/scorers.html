39class HabitatOccupancyScorer(Scorer):
40    """Measures the share of aquatic-capable habitat cells that are
41    actually occupied by adults in the final snapshot.
42
43    This is the field "larval-habitat occupancy" fraction reported by
44    longitudinal surveys (Bogh 2004; Ndenga 2011). A score of 1 means
45    the occupancy falls inside the literature band declared in
46    ``OCC_BAND``; outside it the score falls off as a Gaussian around
47    ``OCC_CENTER``, so 0 (no habitat occupied) scores 0.
48    """
49
50    name = "D1_habitat_occupancy"
51    kind = "composite"
52    weight = 1.0
53    description = "Occupied habitat cells vs literature 20-70% (Bogh 2004; Ndenga 2011)"
54    min_score = 0.5
55    hard_floor = 0.2
56
57    def score(self, run_dir: Path, ctx: dict[str, Any]) -> ScorerResult:
58        """Compute the habitat occupancy fraction and its score.
59
60        Returns the occupied share of habitat cells as the value, or a
61        skipped result when the run has no state snapshot or no habitat
62        cells at all.
63        """
64        target = "0.20-0.70 of habitat cells occupied"
65        final = find_final_state(run_dir)
66        if final is None:
67            return ScorerResult.skipped("no state files", target)
68        with rasterio.open(final) as ds:
69            density = ds.read(1)
70            habitat = ds.read(2)
71        valid_d = np.isfinite(density) & (density != NODATA)
72        valid_h = np.isfinite(habitat) & (habitat != NODATA)
73        habitat_cells = valid_h & (habitat > 0)
74        n_habitat = int(habitat_cells.sum())
75        if n_habitat == 0:
76            return ScorerResult.skipped("no habitat cells (band 2 empty)", target)
77        occupied = valid_d & (density > 0) & habitat_cells
78        occupancy = int(occupied.sum()) / n_habitat
79        lo, hi = OCC_BAND
80        if lo <= occupancy <= hi:
81            score = 1.0
82        else:
83            score = max(0.0, math.exp(-((occupancy - OCC_CENTER) / OCC_SIGMA) ** 2))
84        return ScorerResult(
85            score=round(score, 4),
86            value=float(occupancy),
87            target=target,
88            diagnostics={
89                "occupied_habitat_cells": int(occupied.sum()),
90                "habitat_cells": n_habitat,
91                "final_file": final.name,
92            },
93            passed=score >= self.min_score,
94        )

class SurvivalScorer(mal_core.scoring.scorers.Scorer): 
21class SurvivalScorer(Scorer):
22    """Measures the mean adult lifespan of the simulated population.
23
24    The lifespan is derived from the mean daily adult survival rate of
25    the steady-state half of the run. The score is 1 when the lifespan
26    sits at the centre of the accepted field window and decays as a
27    Gaussian away from it, so a population that dies out too fast
28    scores near 0.
29    """
30
31    name = "D2_survival"
32    kind = "composite"
33    weight = 4.0
34    description = "Mean adult lifespan vs 8-15 d (Saarman 2019, Midega 2007)"
35    min_score = 0.5
36    hard_floor = 0.2
37
38    def score(self, run_dir: Path, ctx: dict[str, Any]) -> ScorerResult:
39        """Compute the mean adult lifespan in days and its score.
40
41        Returns the lifespan as the value, or a skipped result when the
42        cohort log is missing, unreadable or too short to have a
43        steady-state phase.
44        """
45        cohort_path = find_cohort_file(run_dir)
46        if cohort_path is None:
47            return ScorerResult.skipped(
48                "no cohort file (*_cohort.json)", target="8-15 days"
49            )
50        try:
51            data = json.loads(cohort_path.read_text())
52        except (json.JSONDecodeError, OSError) as e:
53            return ScorerResult.skipped(f"cohort unreadable: {e}", target="8-15 days")
54        daily = data.get("daily", [])
55        if len(daily) < 8:
56            return ScorerResult.skipped(
57                f"cohort too short ({len(daily)} days)", target="8-15 days"
58            )
59        steady = daily[len(daily) // 2 :]
60        rates = [
61            1.0 - (e.get("n_deaths", 0) / e["n_adults"])
62            for e in steady
63            if e.get("n_adults", 0) > 0
64        ]
65        if not rates:
66            return ScorerResult.skipped(
67                "no valid adult survival days", target="8-15 days"
68            )
69        p_d = sum(rates) / len(rates)
70        mean_life = 1.0 / (1.0 - p_d) if p_d < 1.0 else float("inf")
71        score = math.exp(-((mean_life - 11.0) / 4.0) ** 2)
72        return ScorerResult(
73            score=round(score, 4),
74            value=round(mean_life, 2),
75            target="8-15 days",
76            diagnostics={
77                "p_d_mean": round(p_d, 4),
78                "n_steady_days": len(steady),
79            },
80            passed=score >= self.min_score,
81        )

23class EIPScorer(Scorer):
24    """Measures how far the infected mosquito cohort has progressed
25    through the extrinsic incubation period.
26
27    The value is the fraction of infected mosquitoes whose EIP has
28    completed at the measurement day. The score is 1 when that fraction
29    falls inside the accepted window and decays as a Gaussian around
30    the window centre otherwise. Runs without transmission dynamics
31    (a constant fraction) are skipped rather than scored.
32    """
33
34    name = "D3_eip"
35    kind = "composite"
36    weight = 3.0
37    description = "Completed-EIP fraction at day 30 vs 0.20-0.50 (transmission runs)"
38    min_score = 0.3
39    hard_floor = 0.1
40
41    def score(self, run_dir: Path, ctx: dict[str, Any]) -> ScorerResult:
42        """Compute the completed-EIP fraction at the measurement day and its score.
43
44        Returns the completed fraction as the value, or a skipped result
45        when the cohort log is missing or carries no EIP dynamics.
46        """
47        cohort_path = find_cohort_file(run_dir)
48        if cohort_path is None:
49            return ScorerResult.skipped(
50                "no cohort file (*_cohort.json)", target="0.20-0.50"
51            )
52        try:
53            data = json.loads(cohort_path.read_text())
54        except (json.JSONDecodeError, OSError) as e:
55            return ScorerResult.skipped(f"cohort unreadable: {e}", target="0.20-0.50")
56        daily = data.get("daily", [])
57        if not daily:
58            return ScorerResult.skipped("empty cohort", target="0.20-0.50")
59
60        eip_values = [e.get("eip_frac", 0.0) for e in daily]
61        if len(set(eip_values)) <= 1:
62            return ScorerResult.skipped(
63                "eip_frac is constant — transmission dynamics not recorded "
64                "(enable --enable-transmission)",
65                target="0.20-0.50",
66            )
67
68        day30 = next((e for e in daily if e.get("day") == 30), daily[-1])
69        eip_frac = day30.get("eip_frac", 0.0)
70        if 0.20 <= eip_frac <= 0.50:
71            score = 1.0
72        else:
73            score = math.exp(-((eip_frac - 0.35) / 0.15) ** 2)
74        return ScorerResult(
75            score=round(score, 4),
76            value=round(eip_frac, 4),
77            target="0.20-0.50",
78            diagnostics={"day": day30.get("day"), "n_days": len(daily)},
79            passed=score >= self.min_score,
80        )

 56          Day 0-2: rain pulses (recharge; the activation pulse)
 57          Day 3-12: 0 mm (drying memory)
 58

 60          1. z stays above the activation floor for several days

 71            fill = 1.0 - math.exp(-max(0.0, rain) / _PLUVIAL_RAIN_SCALE_MM)

 83                active_days += 1

 87            z_series.append(z)

 89                active_days += 1

 91                dry_day = d + 3
