 37    def _resolve_binary(self) -> Path:
 38        """Locate the compiled ABM binary.
 39
 40        Looks in the worktree (if given) and then in the installed package
 41        directory, preferring the platform-named binary under ``bin/`` over
 42        the generic one under ``build/src/``. Raises FileNotFoundError with
 43        the compile command when no binary is found.
 44        """
 45        candidate_dirs: list[Path] = []
 46        worktree = getattr(self, "worktree", None)
 47        if worktree:
 48            wt_abm = worktree / "mal-core" / "src" / "mal_core" / "abm"
 49            if wt_abm.is_dir():
 50                candidate_dirs.append(wt_abm)
 51            elif (worktree / "CMakeLists.txt").is_file():
 52                candidate_dirs.append(worktree)
 53            else:
 54                candidate_dirs.append(wt_abm)
 55        pkg_dir = Path(__file__).parent
 56        candidate_dirs.append(pkg_dir)
 57
 58        for base in candidate_dirs:
 59            bin_path = base / "bin" / f"mal_abm_fast_{sys.platform}"
 60            if bin_path.exists():
 61                return bin_path
 62            build_path = base / "build" / "src" / "mal_abm_fast"
 63            if build_path.exists():
 64                return build_path
 65
 66        err_hint = "malariasim abm --compile"
 67        if worktree:
 68            err_hint += f" --worktree {worktree}"
 69        raise FileNotFoundError(f"ABM binary not found. Run: {err_hint}")

 71    def _introspect_flags(self) -> dict[str, dict[str, Any]]:
 72        """Parse the binary's ``--help`` output into a flag schema.
 73
 74        Returns a mapping of flag name (underscored) to its type, default
 75        and help text. An empty mapping is returned when the binary cannot
 76        be executed or times out.
 77        """
 78        try:
 79            result = subprocess.run([str(self.binary), "--help"], capture_output=True, text=True)
 80            output = result.stdout + result.stderr
 81        except (subprocess.TimeoutExpired, FileNotFoundError):
 82            return {}
 83        flags: dict[str, dict[str, Any]] = {}
 84        for match in re.finditer(r"--([\w-]+)\s+(?:<(\w+)>)?\s*(.*)", output):
 85            name = match.group(1).replace("-", "_")
 86            ftype = match.group(2) or "bool"
 87            desc = match.group(3).strip()
 88            tmap = {"int": int, "float": float, "str": str, "string": str, "bool": bool}
 89            flags[name] = {"type": tmap.get(ftype, str), "default": None, "help": desc}
 90        return flags

 92    def run(self, _timeout: int | None = None, **flags) -> dict[str, Any]:
 93        """Run the binary in its ``run`` subcommand with the given flags.
 94
 95        Keyword names become dashed command-line flags; ``None`` values are
 96        dropped and booleans are passed only when true.
 97
 98        Returns:
 99            A dict with the captured ``stdout``, ``stderr`` and the process
100            ``returncode``.
101        """
102        if self._flags_schema is None:
103            self._flags_schema = self._introspect_flags()
104        cmd = [str(self.binary), "run"]
105        for name, value in flags.items():
106            if value is None:
107                continue
108            cli = f"--{name.replace('_', '-')}"
109            if isinstance(value, bool):
110                if value:
111                    cmd.append(cli)
112            else:
113                cmd.extend([cli, str(value)])
114        result = subprocess.run(cmd, capture_output=True, text=True, timeout=_timeout, check=False)
115        return {"stdout": result.stdout, "stderr": result.stderr, "returncode": result.returncode}

117    def get_flags_schema(self) -> dict[str, dict[str, Any]]:
118        """Return the flag schema of the binary, introspecting it once."""
119        if self._flags_schema is None:
120            self._flags_schema = self._introspect_flags()
121        return self._flags_schema

259    # Native in-process GIF (plan 04). `gif` arrives in three shapes:
260    #   None/False   -> no GIF
261    #   ""           -> bare `--gif`: default `<output_dir>/<aoi>_abm_seedNNNN.gif`
262    #   str/Path     -> explicit output path, forwarded verbatim
263    # The render knobs are forwarded to the C++ CLI in all enabled cases.
264    #
265    # TRANSMISSION GIF (plan 05): when the run has enable_transmission, the
266    # C++ binary ALSO emits a second, automatic GIF named
267    # `<stem>_transmission.gif` (sibling of the habitat GIF, or of `--output`
268    # when `--gif` is absent). It shares --gif-every/--gif-fps/--gif-max-width
269    # and needs NO new kwarg here — its path is computed inside the binary.
270    # The binary writes the structured sibling `<stem>_transmission_debug.jsonl`
271    # automatically as well; each rollout gets its own seed-suffixed artifact.
272    gif = kwargs.pop("gif", None)
273    if gif is not None and gif is not False:
274        gif_path = (
275            Path(gif)
276            if isinstance(gif, (str, Path)) and str(gif) != ""
277            else output_dir / f"{aoi}_abm_seed{seed:04d}.gif"
278        )
279        flags["gif"] = str(gif_path)
280    for key in ("gif_every", "gif_fps", "gif_max_width"):
281        value = kwargs.pop(key, None)
282        if value is not None:
283            flags[key] = value
284
285    flags.update(kwargs)
286
287    log.info("Running ABM for %s (year=%d, seed=%d)", aoi, year, seed)
288    wrapper = CppAbmWrapper(worktree=worktree)
289    result = wrapper.run(**flags, _timeout=timeout)
290
291    return {"output_path": str(output_path), **result}

Run ABM using manifest to resolve data paths.

Reads manifest, validates completeness, resolves paths, then calls the C++ binary via CppAbmWrapper.

When enable_transmission is true, the binary also writes a versioned per-run JSONL debug artifact beside its automatic transmission GIF: <transmission-GIF-stem>_debug.jsonl (for the default output this is <aoi>_abm_seedNNNN_transmission_debug.jsonl). It contains one metadata record and one daily record with compartment counts, bites by host group, and vector seeding/exposure/EIP event counts. This is distinct from the legacy emit_transmission_log daily summary JSON.
